Two phases: Phase 1 removes dead exports and extracts the bridge-test boilerplate into a shared helper (identified by the v0.4.5 pre-flight refactor analysis); Phase 2 installs the `bench/` directory with Vitest's native bench API and records baseline timings for the three performance-critical paths in the codebase.

The one deliberate call already made by Daniel: `Float64Tensor` is `@internal`, and its `export` keyword will be removed; the class has never appeared in `src/index.ts` and is not in the public surface, so this is non-breaking. Phase 2 reuses the existing Vitest 4.1.4 devDependency (`vitest bench`) — no new devDependency is needed. Bench files live under `bench/*.bench.ts` and are excluded from the normal test pass by the existing `vitest.config.ts` `include` pattern.

**Spec:** `docs/architecture/v0.4.5-refactor-targets.md` (generated 2026-05-16 by direct code inspection). All 10 tasks derive from that analysis — every verdict was independently verified against source and test files; static-analyzer output from the earlier unused-analysis pass was the starting checklist only.

```typescript
/**
 * Shared test helpers for bridge equation encoding tests.
 *
 * v0.4.5: extracted from 39 encoding test files to eliminate the
 * repeated 3-part boilerplate (catalog lookup + dimensional validation
 * + round-trip pin). Net savings: ~295 LOC across the test suite.
 *
 * NOT for use in reformulation tests (different structure) or fix tests
 * (non-standard shapes). Encoding tests only.
 *
 * @module tests/bridges/_helpers
 */
import { expect } from 'vitest';
import { BRIDGE_EQUATIONS, BridgeEquationStatus } from '../../src/bridges/index.js';
import type { ExprNode } from '../../src/dimensional/validator.js';
import { validate } from '../../src/dimensional/validator.js';
import { format } from '../../src/dimensional/algebra.js';

/**
 * Assert that a bridge with the given id exists in BRIDGE_EQUATIONS, and
 * optionally that its status matches the expected value.
 *
 * Usage:
 *   const be22 = expectBridgeInIndex(22, 'speculative');
 *   // returns the BridgeEquationEntry for further assertions
 */
export function expectBridgeInIndex(
  id: number,
  status?: BridgeEquationStatus,
) {
  const entry = BRIDGE_EQUATIONS.find((e) => e.id === id);
  expect(entry, `Bridge BE-${id} not found in BRIDGE_EQUATIONS`).toBeDefined();
  if (status !== undefined) {
    expect(entry!.status, `BE-${id} status mismatch`).toBe(status);
  }
  return entry!;
}

/**
 * Assert that the given AST node validates cleanly and its inferred
 * SI dimensional signature matches the expected string.
 *
 * Usage:
 *   expectDimRoundTrip(BE22_TOPOLOGICAL_ENTANGLEMENT_RHS, '[1]');
 */
export function expectDimRoundTrip(rhs: ExprNode, expectedSig: string): void {
  const r = validate(rhs);
  expect(r.ok, `AST validation failed: ${JSON.stringify(r.violations)}`).toBe(true);
```

```typescript
/**
 * AD forward + reverse mode benchmark.
 *
 * Function: fn(x) = x * x  (element-wise self-multiplication)
 * Engines: Float64ReferenceEngine (always), MathTSEngine (optional — skipped
 *   if @danielsimonjr/mathts-tensor or @danielsimonjr/mathts-autograd is absent)
 * Shapes: [10], [100], [10, 10], [100, 100]
 *
 * v0.4.5: establishes AD performance baselines. No threshold gates.
 *
 * Skipping MathTSEngine: vitest bench does not have a native skip() for
 * bench() (unlike it.skip). Use a wrapper that no-ops if the engine is
 * unavailable, logging a one-time warning.
 */
import { bench, describe, beforeAll } from 'vitest';
import { Float64ReferenceEngine } from '../src/numerical/float64-engine.js';
import { hasAutogradSupport } from '../src/numerical/index.js';
import type { TensorEngine } from '../src/numerical/tensor-engine.js';

// ---------------------------------------------------------------------------
// Engine setup
// ---------------------------------------------------------------------------

const f64 = new Float64ReferenceEngine();

let mathts: TensorEngine | null = null;
let mathtsAvailable = false;

// Attempt to load MathTSEngine (optional dependency).
try {
  const { MathTSEngine } = await import('../src/numerical/mathts-engine.js');
  mathts = new MathTSEngine();
  mathtsAvailable = hasAutogradSupport(mathts);
} catch {
  // @danielsimonjr/mathts-tensor or @danielsimonjr/mathts-autograd not installed.
  console.warn('[bench/ad] MathTSEngine unavailable — skipping MathTS AD benchmarks.');
}

// ---------------------------------------------------------------------------
// Helper: build a flat NestedArray of the given shape filled with 1.0
// ---------------------------------------------------------------------------
function ones(shape: number[]): number | number[] | number[][] {
  if (shape.length === 0) return 1.0;
  if (shape.length === 1) return Array.from({ length: shape[0] }, () => 1.0);
  return Array.from({ length: shape[0] }, () =>
    ones(shape.slice(1)) as number[],
  );
}

// ---------------------------------------------------------------------------
// Bench suites — one per shape
// ---------------------------------------------------------------------------

for (const shape of [[10], [100], [10, 10], [100, 100]] as number[][]) {
  const label = `[${shape.join('×')}]`;
  const xData = ones(shape);

  describe(`AD — shape ${label}`, () => {
    bench(`Float64 forward fn=x*x shape=${label}`, async () => {
      const x = await f64.fromNested(shape, xData as never);
      await f64.forwardGrad!((t) => f64.mul(t, t), x);
    });

    bench(`Float64 reverse fn=x*x shape=${label}`, async () => {
      const x = await f64.fromNested(shape, xData as never);
      await f64.reverseGrad!((t) => f64.mul(t, t), x);
    });

    if (mathtsAvailable && mathts !== null) {
      const mt = mathts;
      bench(`MathTS forward fn=x*x shape=${label}`, async () => {
        const x = await mt.fromNested(shape, xData as never);
        await mt.forwardGrad!((t) => mt.mul(t, t), x);
      });

      bench(`MathTS reverse fn=x*x shape=${label}`, async () => {
        const x = await mt.fromNested(shape, xData as never);
        await mt.reverseGrad!((t) => mt.mul(t, t), x);
      });
    }
  });
}
```
